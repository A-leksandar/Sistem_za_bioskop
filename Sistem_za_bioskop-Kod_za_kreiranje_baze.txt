create table KORISNIK(
	idKorisnika int identity primary key not null,
	email varchar(100) unique not null,
	lozinka nvarchar(50) not null,
	idOsobe int not null
)
go

alter table KORISNIK
add constraint fk_osoba_korisnik foreign key (idOsobe) references OSOBA (idOsobe)
go

create table KARTA(
	idKarte int identity primary key not null,
	idFilma int not null,
	idSale int not null,
	idSedista int not null,
	idProjekcije int not null
)
go

alter table KARTA
add constraint fk_film_karta foreign key (idFilma) references FILM (idFilma)
go

alter table KARTA
add constraint fk_sala_karta foreign key (idSale) references SALA (idSale)
go

alter table KARTA
add constraint fk_sediste_karta foreign key (idSedista) references SEDISTE (idSedista)
go

alter table KARTA
add constraint fk_projekcija_karta foreign key (idProjekcije) references PROJEKCIJA (idProjekcije)
go

create table TRANSAKCIJA(
	idTransakcije int identity primary key not null,
	brojKartice varchar(16) not null,
	bezbednosniBroj varchar(4) not null,
	idKorisnika int not null,
	idKarte int not null
)
go

alter table TRANSAKCIJA
add constraint fk_korisnik_transakcija foreign key (idKorisnika) references KORISNIK (idKorisnika)
go

alter table TRANSAKCIJA
add constraint fk_karta_transakcija foreign key (idKarte) references KARTA (idKarte)
